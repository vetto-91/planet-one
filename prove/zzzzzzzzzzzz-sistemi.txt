ia off

aspetta 2

aspetta 1

incantesimo calma
aspetta 1.5
verifica /magia < 35
god on
# Silvia diventa compagna dopo la Prova dell'Arco.
missioni on

aspetta 1

aspetta 1
attendi npc:silvia:modo = Ready 90
davanti silvia
aspetta 0.5
guarda silvia
aspetta 1
premi e
aspetta 1
avanza
verifica /dialogo/scelte contiene Vieni con me

aspetta 1
verifica /personaggio/compagno = Silvia
tp 30 3 -20
aspetta 6
verifica npc:silvia:dist < 6
camera 200 20 8
aspetta 1

# Un colpo: salvata e ricaricata, Silvia ti segue ancora con la vita che aveva.

aspetta 1

premi f5
aspetta 2
premi f9
aspetta 10
verifica /personaggio/compagno = Silvia

verifica /personaggio/compagno_vita < 35

aspetta 1
verifica /personaggio/compagno != Silvia
verifica /personaggio/feriti = Silvia
